## 📋 实施步骤

### 1. ViewModel 增强（`Sources/Session/SessionReviewViewModel.swift`）

```swift
@MainActor
final class SessionReviewViewModel: ObservableObject {
    @Published var utterances: [UtteranceItem] = []
    @Published var eval: EvalSummary?
    @Published var audioPlayback: AudioPlaybackState = .idle
    
    private let api: SessionAPIClient
    
    func load(sessionID: String) async {
        do {
            let review = try await api.getReview(sessionID: sessionID)
            utterances = review.utterances
            eval = review.eval
        } catch {
            // 错误处理
        }
    }
    
    func playAudio(utteranceID: String) async {
        audioPlayback = .loading
        do {
            let audioURL = try await api.getAudioURL(sessionID: currentSessionID, utteranceID: utteranceID)
            audioPlayback = .playing(url: audioURL)
        } catch {
            audioPlayback = .error
        }
    }
}

struct UtteranceItem: Codable, Identifiable {
    let id: String
    let speaker: String  // "user" / "ai"
    let text: String
    let audioURL: String?  // 仅 user 有
    let startedAtMs: Int
    let endedAtMs: Int
}

struct EvalSummary: Codable {
    let score: Double
    let dims: [String: Double]  // grammar / fluency / vocabulary
    let suggestions: [String]
}
```

### 2. View 增强（`Sources/Session/SessionReviewView.swift`）

```swift
var body: some View {
    ScrollView {
        LazyVStack(spacing: 12) {
            ForEach(viewModel.utterances) { utt in
                UtteranceRow(utterance: utt, onPlayAudio: {
                    Task { await viewModel.playAudio(utteranceID: utt.id) }
                })
            }
            
            if let eval = viewModel.eval {
                EvalCard(eval: eval)
            }
        }
        .padding()
    }
    .navigationTitle("会话回顾")
}
```

### 3. `UtteranceRow` UI

- **AI utterance**：左侧气泡，浅蓝底色
- **User utterance**：右侧气泡，白底色 + 「▶」播放按钮
- 点击播放按钮 → 调 `playAudio` + 进度条（10s 内）
- 播放中切换到其他 utterance → 自动停止上一段

### 4. `EvalCard` UI（3 维可视化）

```swift
struct EvalCard: View {
    let eval: EvalSummary
    var body: some View {
        VStack(alignment: .leading) {
            Text("总评 \(String(format: "%.0f", eval.score * 100))").font(.title3)
            ForEach(eval.dims.sorted(by: { $0.key < $1.key }), id: \.key) { dim in
                HStack {
                    Text(dimLabel(dim.key)).frame(width: 80, alignment: .leading)
                    ProgressView(value: dim.value).frame(maxWidth: .infinity)
                    Text("\(Int(dim.value * 100))").frame(width: 40, alignment: .trailing)
                }
            }
            if !eval.suggestions.isEmpty {
                Divider()
                Text("改进建议").font(.subheadline.bold())
                ForEach(eval.suggestions, id: \.self) { s in
                    Text("• \(s)")
                }
            }
        }
        .padding()
        .background(Color(.systemGroupedBackground))
        .clipShape(RoundedRectangle(cornerRadius: 12))
    }
}
```
