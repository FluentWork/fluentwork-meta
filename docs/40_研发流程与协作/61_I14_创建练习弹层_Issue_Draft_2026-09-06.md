### 2. ViewModel `Sources/CreatePracticeViewModel.swift`

```swift
@MainActor
final class CreatePracticeViewModel: ObservableObject {
    @Published var refineStatus: RefineStatus = .idle
    @Published var error: APIError?
    private let api: MaterialsAPIClient
    private var pollTask: Task<Void, Never>?
    
    enum RefineStatus { case idle, queued, processing, ready, failed }
    
    func submit(text: String, kind: MaterialKind) async {
        do {
            let materialID = try await api.createMaterial(kind: kind, content: text)
            refineStatus = .queued
            startPolling(materialID: materialID)
        } catch {
            self.error = error as? APIError
        }
    }
    
    private func startPolling(materialID: String) {
        pollTask?.cancel()
        pollTask = Task {
            while !Task.isCancelled {
                try? await Task.sleep(nanoseconds: 2_000_000_000) // 2s
                if Task.isCancelled { return }
                let mat = try? await api.getMaterial(id: materialID)
                guard let mat else { continue }
                await MainActor.run { self.refineStatus = RefineStatus(rawValue: mat.refineStatus) ?? .processing }
                if mat.refineStatus == "ready" || mat.refineStatus == "failed" { return }
            }
        }
    }
}
```
