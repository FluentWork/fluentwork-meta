### 1.4 语义判定 Prompt（D-3 拍板：Ark Mini）

```yaml
system: |
  You are a strict semantic-equivalence judge for English phrases in workplace contexts.
  Given a TARGET phrase and a USER_SAID phrase, decide whether they convey the same intent.
  Respond ONLY with JSON: {"pass": true|false, "score": 0.0-1.0, "reason": "<20 chars zh>"}

user: |
  TARGET: {{target.chunk_en}}
  INTENT_ZH: {{target.intent_zh}}
  USER_SAID: {{user.asr_text}}
  CONTEXT: {{target.scene_tag}} / {{target.function_tag}}
```

LLM 响应解析后落 `drill_records` 表 + 更新 `phrase_blocks.state`（D-4 SM-2）：
- 失败 → 插入本轮尾部 + `next_due_at = now + 1h`
- 成功 + `success_streak < 3` → `success_streak += 1`, `next_due_at = now + 24h`
- 成功 + `success_streak >= 3` → `state = 'automated'`, `next_due_at = now + 7d`
- `state = 'automated'` 成功 → `next_due_at = now + 30d`（可配置上限）

### 2.1 创建迁移 0012（49_ §五）

```sql
-- 0012_alter_privacy_soft_delete.sql
ALTER TABLE users             ADD COLUMN deleted_at    DATETIME(3) NULL,
                             ADD COLUMN tombstone_at DATETIME(3) NULL;
ALTER TABLE materials         ADD COLUMN deleted_at    DATETIME(3) NULL;
ALTER TABLE practice_sessions ADD COLUMN deleted_at    DATETIME(3) NULL;
ALTER TABLE reviews           ADD COLUMN deleted_at    DATETIME(3) NULL;
ALTER TABLE ai_cost_logs      ADD COLUMN user_id_anonymized VARCHAR(64) NULL;

CREATE TABLE IF NOT EXISTS tombstones (
    id CHAR(36) PRIMARY KEY,
    user_id CHAR(36) NOT NULL,
    entity_type VARCHAR(32) NOT NULL,
    entity_id CHAR(36) NOT NULL,
    tombstoned_at DATETIME(3) NOT NULL,
    INDEX idx_tomb_user (user_id, tombstoned_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
  COMMENT 'A4 Outbox 同步 tombstone (45 §2.2)';
```

### 2.2 service 层（`internal/account/privacy_service.go`）

```go
type PrivacyService struct {
    store      PrivacyStore
    outbox     OutboxClient   // 写 tombstones
    logger     *slog.Logger
}

type DeleteResult struct {
    DeletedAt     time.Time         `json:"deleted_at"`
    Cascaded      map[string]int    `json:"cascaded"`
    BackupPurgeAt time.Time         `json:"backup_purge_at"`  // = deleted_at + 30d
}

func (s *PrivacyService) DeleteAllData(ctx context.Context, userID, confirmationCode string) (*DeleteResult, error) {
    // 1. 校验 confirmation_code == "DELETE-MY-DATA"，否则 422
    // 2. 单事务：
    //    - UPDATE users SET deleted_at = NOW(), tombstone_at = NOW() WHERE id = ?
    //    - UPDATE materials SET deleted_at = NOW() WHERE user_id = ?
    //    - UPDATE practice_sessions SET deleted_at = NOW() WHERE user_id = ?
    //    - UPDATE reviews SET deleted_at = NOW() WHERE user_id = ?
    //    - UPDATE phrase_blocks SET deleted_at = NOW() WHERE user_id = ?  (V2.0 增量)
    //    - DELETE FROM drill_records WHERE user_id = ?  (物理删除)
    //    - UPDATE ai_cost_logs SET user_id_anonymized = 'deleted_user_xxx' WHERE user_id = ?
    //    - INSERT INTO tombstones (...) VALUES (...), (...), ...
    // 3. 幂等：检测 users.deleted_at 已存在则返回 cascaded: all_zeros + 200
}
```
