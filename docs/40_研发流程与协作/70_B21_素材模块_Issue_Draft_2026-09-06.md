```go
type Service struct {
    store      Store
    refiner    Refiner

}

func (s *Service) Create(ctx context.Context, userID, kind, content string) (string, error) {
    // 1. URL 类型 → 调用 URL fetcher 抓取正文（V1.5 接入；本期 stub）
    if kind == "url" {
        // TODO B16: URLFetcher.Fetch(url)
        content = "[URL content placeholder]"
    }
    
    // 2. 校验长度（文本 ≤ 5000 字符；URL ≤ 500 字符）
    if len(content) > 5000 {
        return "", apierr.BadRequest("content_too_long")
    }
    
    // 3. 落库 material（refine_status=queued）
    materialID, err := s.store.InsertMaterial(ctx, userID, kind, content)
    if err != nil { return "", err }
    
    // 4. 异步入队 RefineJob
    if err := s.job.Enqueue(RefineJob{MaterialID: materialID, UserID: userID}); err != nil {
        s.logger.Warn("enqueue refine failed", "material_id", materialID, "err", err)
    }
    
    return materialID, nil
}

func (s *Service) Get(ctx context.Context, userID, materialID string) (*Material, error) {
    return s.store.GetMaterial(ctx, userID, materialID)
}
```

### 4. Refiner（提炼流水线，`refiner.go`）

```go
type Refiner struct {
    llm     LLMClient
    blocks  PhraseBlockStore  // 写入 phrase_blocks 表
    logger  *slog.Logger
}

func (r *Refiner) Refine(ctx context.Context, materialID string) (*RefineResult, error) {
    mat, err := r.store.GetMaterial(ctx, materialID)
    if err != nil { return nil, err }
    
    // 1. 构造提炼 Prompt（输入原文 → 输出 5-20 个 phrase_blocks JSON 数组）
    prompt := BuildRefinePrompt(mat.Content, mat.Kind)
    
    raw, err := r.llm.Complete(ctx, LLMRequest{
        Model: "ark-mini",
        SystemPrompt: prompt.System,
        UserPrompt:   prompt.User,
        Temperature:  0.5,
        MaxTokens:    3000,
        Timeout:      30 * time.Second,  // 提炼比 eval 更宽容
    })
    if err != nil {
        r.markFailed(ctx, materialID, "llm_timeout")
        return nil, err
    }
    
    // 2. 解析 LLM 输出
    var blocks []RefinedBlock
    if err := json.Unmarshal([]byte(raw), &blocks); err != nil {
        r.markFailed(ctx, materialID, "parse_error")
        return nil, err
    }
    
    // 3. 落 phrase_blocks（关联 material_id + user_id）
    blockIDs, err := r.blocks.InsertBulk(ctx, materialID, mat.UserID, blocks)
    if err != nil {
        r.markFailed(ctx, materialID, "db_error")
        return nil, err
    }
    
    // 4. 更新 material.refine_status=ready + block_count
    if err := r.store.MarkRefined(ctx, materialID, len(blockIDs)); err != nil {
        return nil, err
    }
    
    return &RefineResult{MaterialID: materialID, BlockIDs: blockIDs, BlockCount: len(blockIDs)}, nil
}
```

### 5. 提炼 Prompt 模板（`refiner_prompt.go`）

```yaml
system: |
  You are a workplace English phrase extractor.
  Given a source text, extract 5-20 useful English phrases (chunks) that a working professional
  would want to learn. For each phrase, provide:
  - chunk_en: the exact English phrase (5-30 words)
  - intent_zh: what the phrase means in Chinese (≤ 15 chars)
  - scene_tag: workplace scenario (meeting, presentation, email, chat, negotiation, customer-support)
  - function_tag: speech function (request, agree, disagree, propose, clarify, summarize, negotiate, ask)
  - difficulty: 1 (easy) / 2 (medium) / 3 (hard)
  Output ONLY a JSON array. No markdown. No commentary.

user: |
  SOURCE_KIND: {{mat.kind}}
  SOURCE_CONTENT:
  {{mat.content}}
```

### 6. HTTP 路由（48 §1.2）

```go
// internal/materials/http.go
func RegisterRoutes(rg gin.IRouter, h *Handler) {
    rg.POST("/materials",  h.accounts.RequireAuth(), h.PostMaterial)
    rg.GET("/materials/:id", h.accounts.RequireAuth(), h.GetMaterial)
}
```

```go
func (h *Handler) PostMaterial(c *gin.Context) {
    var req struct {
        Kind    string `json:"kind"`     // "text" | "voice_note" | "url"
        Content string `json:"content"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        httpjson.Error(c, apierr.BadRequest("invalid_body"))
        return
    }
    
    materialID, err := h.svc.Create(c, h.accounts.UserID(c), req.Kind, req.Content)
    if err != nil {
        httpjson.Error(c, err)
        return
    }
    
    httpjson.Accepted(c, gin.H{
        "material_id":   materialID,
        "refine_status": "queued",
    })
}

func (h *Handler) GetMaterial(c *gin.Context) {
    materialID := c.Param("id")
    mat, err := h.svc.Get(c, h.accounts.UserID(c), materialID)
    if err != nil {
        httpjson.Error(c, err)
        return
    }
    httpjson.OK(c, mat)
}
```
