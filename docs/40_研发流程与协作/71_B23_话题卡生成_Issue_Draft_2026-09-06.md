### 3. Service 层（`service.go`）

```go
type Service struct {
    store      Store
    generator  Generator
    job        JobQueue
    logger     *slog.Logger
}

func (s *Service) ListToday(ctx context.Context, userID string) ([]TopicCard, error) {
    return s.store.ListTodayCards(ctx, userID, time.Now())
}

func (s *Service) Checkin(ctx context.Context, userID, cardID string, reflection string) (*CheckinResult, error) {
    // 1. 校验 card 归属
    card, err := s.store.GetCard(ctx, cardID)
    if err != nil { return nil, err }
    if card.UserID != userID { return nil, apierr.Forbidden("not_owner") }
    if card.CheckinAt != nil { return nil, apierr.Conflict("already_checked_in") }
    
    // 2. 写 checkin_at + reflection
    now := time.Now()
    if err := s.store.MarkCheckedIn(ctx, cardID, now, reflection); err != nil {
        return nil, err
    }
    
    // 3. 更新 streak
    streak, err := s.streakStore.UpdateOnCheckin(ctx, userID, now)
    if err != nil {
        s.logger.Warn("streak update failed", "user_id", userID, "err", err)
    }
    
    // 4. 写 checkins 表（V2.0 新表，记录复盘文本）
    checkinID, err := s.store.InsertCheckin(ctx, cardID, userID, reflection, now)
    if err != nil {
        s.logger.Warn("checkin insert failed", "user_id", userID, "err", err)
    }
    
    return &CheckinResult{
        CheckinID:  checkinID,
        StreakDays: streak.CurrentStreak,
    }, nil
}
```

### 4. Generator（LLM 生成，`generator.go`）

```go
type Generator struct {
    llm      LLMClient
    analytics AnalyticsStore  // 用户标签分布查询
    logger    *slog.Logger
}

func (g *Generator) GenerateForUser(ctx context.Context, userID string, targetDate time.Time) ([]TopicCard, error) {
    // 1. 拉取用户最近 30 天练习标签分布
    distribution, err := g.analytics.GetTagDistribution(ctx, userID, 30)
    if err != nil {
        return nil, err
    }
    
    // 2. 拉取用户当前学习水平
    level, err := g.analytics.GetUserLevel(ctx, userID)
    if err != nil {
        return nil, err
    }
    
    // 3. 拉取历史话题卡（避免重复）
    recentTitles, err := g.store.GetRecentCardTitles(ctx, userID, 14)
    if err != nil {
        return nil, err
    }
    
    // 4. 构造生成 Prompt，输出 3 张话题卡 JSON
    prompt := BuildGeneratorPrompt(distribution, level, recentTitles)
    
    raw, err := g.llm.Complete(ctx, LLMRequest{
        Model:        "ark-mini",
        SystemPrompt: prompt.System,
        UserPrompt:   prompt.User,
        Temperature:  0.7,  // 比 eval 更富创造性
        MaxTokens:    1500,
        Timeout:      15 * time.Second,
    })
    if err != nil {
        return nil, err
    }
    
    // 5. 解析 + 落库
    var cards []TopicCard
    if err := json.Unmarshal([]byte(raw), &cards); err != nil {
        return nil, fmt.Errorf("topic_card_parse: %w", err)
    }
    
    validUntil := targetDate.Add(24 * time.Hour)
    return g.store.InsertCards(ctx, userID, cards, validUntil, targetDate)
}
```

### 6. Scheduler（`scheduler.go`）— 复用 D-5 cron worker

```go
// 在 D-5 现有的 daily-broadcast worker 中追加 task
func (s *Scheduler) DailyTopicCardGeneration(ctx context.Context, runDate time.Time) error {
    // 1. 拉取活跃用户（最近 30 天有 practice_session）
    activeUsers, err := s.analytics.GetActiveUsers(ctx, 30)
    if err != nil { return err }
    
    s.logger.Info("topic_card_generation_started", "user_count", len(activeUsers), "date", runDate)
    
    // 2. 并发生成（限速 10 QPS 防 rate limit）
    sem := make(chan struct{}, 10)
    var wg sync.WaitGroup
    for _, userID := range activeUsers {
        sem <- struct{}{}
        wg.Add(1)
        go func(uid string) {
            defer wg.Done()
            defer func() { <-sem }()
            cards, err := s.generator.GenerateForUser(ctx, uid, runDate)
            if err != nil {
                s.logger.Warn("topic_card_gen_failed", "user_id", uid, "err", err)
                return
            }
            s.logger.Info("topic_card_gen_done", "user_id", uid, "card_count", len(cards))
        }(userID)
    }
    wg.Wait()
    
    return nil
}
```

### 7. HTTP 路由（48 §1.7）

```go
// internal/topic/http.go
func RegisterRoutes(rg gin.IRouter, h *Handler) {
    rg.GET("/topic-cards",            h.accounts.RequireAuth(), h.GetTopicCards)
    rg.POST("/topic-cards/:id/checkin", h.accounts.RequireAuth(), h.PostCheckin)
}
```

```go
func (h *Handler) GetTopicCards(c *gin.Context) {
    userID := h.accounts.UserID(c)
    cards, err := h.svc.ListToday(c, userID)
    if err != nil { httpjson.Error(c, err); return }
    httpjson.OK(c, gin.H{"items": cards})
}

func (h *Handler) PostCheckin(c *gin.Context) {
    var req struct {
        Reflection string `json:"reflection"`
    }
    cardID := c.Param("id")
    result, err := h.svc.Checkin(c, h.accounts.UserID(c), cardID, req.Reflection)
    if err != nil { httpjson.Error(c, err); return }
    httpjson.OK(c, result)
}
```
